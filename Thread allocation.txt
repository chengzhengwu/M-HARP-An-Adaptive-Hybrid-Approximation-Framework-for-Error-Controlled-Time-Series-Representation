package io.github.xkitsios;

import java.lang.management.ManagementFactory;
import java.util.concurrent.*;
import java.util.concurrent.atomic.AtomicLong;


import java.lang.management.OperatingSystemMXBean;

public final class FastThreadPoolUtils {
    /**

     */
    private static Double getSystemLoadAverage() {
        OperatingSystemMXBean osBean = ManagementFactory.getOperatingSystemMXBean();
        double load = osBean.getSystemLoadAverage();
        return load < 0 ? null : load;
    }

    /**

     *
     * @param dataSize       
     * @param epsilon          
    public static void initializeThreadPoolBasedOnDataSize(int dataSize, double epsilon) {
        // 基于数据量确定基础理想线程数
        int baseThreads;
        if (dataSize <= 400_000) {
            baseThreads = 2;
        } else if (dataSize <= 800_000) {
            baseThreads = 4;
        } else if (dataSize <= 1_000_000) {
            baseThreads = 6;
        } else {
            baseThreads = 8; // 
        }


        double epsilonFactor;
        if (epsilon < 0.001) { // 极低 Epsilon（如 <0.1%，如 0.0005=0.05%）
            epsilonFactor = 1.5; // 
        } else if (epsilon < 0.01) { // 低 Epsilon（如 0.1%-1%，如 0.005=0.5%）
            epsilonFactor = 1.0; /
        } else if (epsilon <= 0.05) { // 中等 Epsilon（如 1%-5%）
            epsilonFactor = 0.75; //
        } else { // 高 Epsilon（如 >5%）
            epsilonFactor = 0.5; // 
        }
        int adjustedThreads = (int) Math.round(baseThreads * epsilonFactor);


        // 获取系统可用核心数
        int availableCores = Runtime.getRuntime().availableProcessors();

        // 获取系统负载
        Double systemLoad = getSystemLoadAverage();
        if (systemLoad != null) {

            if (systemLoad > availableCores * 0.7) {
                adjustedThreads = Math.max(1, adjustedThreads / 2);
                System.out.println("High system load detected (" + systemLoad + "), reducing thread count.");
            }
        } else {
            System.out.println("System load average not supported on this platform, skipping load-based adjustment.");
        }


        int optimalThreads = Math.min(adjustedThreads, availableCores);


        ThreadPoolExecutor executor = new ThreadPoolExecutor(
                optimalThreads,
                optimalThreads,
                60L, TimeUnit.SECONDS,
                new LinkedBlockingQueue<>(),
                Executors.defaultThreadFactory(),
                new ThreadPoolExecutor.AbortPolicy());

        FastThreadPoolManager.setCommonExecutor(executor);
        System.out.println("Initialized thread pool with " + optimalThreads
                + " threads (dataSize: " + dataSize
                + ", epsilon: " + epsilon + "%, "
                + "available cores: " + availableCores
                + (systemLoad != null ? ", system load: " + systemLoad : "") + ").");
    }

    /**
     * 关闭线程池，并等待任务完成
     */
    public static void shutdownThreadPool() {
        ExecutorService executor = FastThreadPoolManager.getCommonExecutor();
        if (executor == null) {
            System.out.println("Thread pool is not initialized.");
            return;
        }
        executor.shutdown();
        try {
            if (!executor.awaitTermination(10, TimeUnit.SECONDS)) {
                executor.shutdownNow();
            }
        } catch (InterruptedException e) {
            executor.shutdownNow();
            Thread.currentThread().interrupt();
        }
        System.out.println("Thread pool shut down.");
    }
}
